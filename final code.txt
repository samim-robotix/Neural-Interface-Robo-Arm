// ================================================================
//                         SETUP
// ================================================================
void setup() {
  Serial.begin(115200);

  // --- Initialize NeoPixels ---
  strip.begin();
  strip.setBrightness(255);
  strip.clear();
  strip.show();

  // --- Start Wi-Fi Access Point ---
  Serial.print("Creating Access Point: ");
  Serial.println(hotspot_ssid);
  WiFi.softAP(hotspot_ssid, hotspot_password);
  IPAddress IP = WiFi.softAPIP();
  Serial.print("AP IP address: ");
  Serial.println(IP);

  // --- Define Web Server Routes ---
  server.on("/", HTTP_GET, [](AsyncWebServerRequest *request){
    Serial.println(" -> Request: / (Serving HTML page)");
    request->send_P(200, "text/html", HTML_CONTENT);
  });

  server.on("/toggle", HTTP_GET, [](AsyncWebServerRequest *request){
    isRunning = !isRunning;
    Serial.printf(" -> Request: /toggle. Stimulus is now %s\n", isRunning ? "ON" : "OFF");
    currentStep = 0;
    lastStateChangeTime = millis();
    currentState = WAITING;
    strip.clear(); strip.show();
    request->send(200, "text/plain", "Toggled");
  });

  server.on("/update", HTTP_GET, [](AsyncWebServerRequest *request){
    if (isRunning) {
      Serial.println(" -> Request: /update (Denied: Stimulus is running)");
      request->send(403, "text/plain", "Stop stimulus to update parameters");
      return;
    }
    Serial.println(" -> Request: /update (Updating parameters)");
    if (request->hasParam("color")) {
      String hex = request->getParam("color")->value();
      hex.replace("#", "");
      long number = strtol(hex.c_str(), NULL, 16);
      flickerColor = strip.Color((number >> 16) & 0xFF, (number >> 8) & 0xFF, number & 0xFF);
    }
    if (request->hasParam("hz")) flickerHz = request->getParam("hz")->value().toInt();
    if (request->hasParam("duration")) flickerDurationMs = request->getParam("duration")->value().toFloat() * 1000;
    if (request->hasParam("interval")) intervalMs = request->getParam("interval")->value().toFloat() * 1000;

    Serial.printf("    New Params: %d Hz, %dms duration, %dms interval\n", flickerHz, flickerDurationMs, intervalMs);
    request->send(200, "text/plain", "Updated");
  });
  
  // --- Start server ---
  server.begin();
}

// ================================================================
//                     MAIN LOOP
// ================================================================
void loop() {
  // The server runs in the background. The loop is only for the animation.
  runAnimation();
  // We add a tiny delay to be friendly to the FreeRTOS scheduler
  delay(1);
}

// ================================================================
//                    ANIMATION LOGIC
// ================================================================
void runAnimation() {
  if (!isRunning) {
    strip.clear(); strip.show(); currentStep = 0; currentState = WAITING; return;
  }
  unsigned long currentTime = millis();
  if (currentState == WAITING) {
    if (currentTime - lastStateChangeTime >= intervalMs) {
      currentState = FLICKERING; lastStateChangeTime = currentTime;
    }
  } else if (currentState == FLICKERING) {
    if (currentTime - lastStateChangeTime >= flickerDurationMs) {
      currentStep = (currentStep + 1) % 6;
      currentState = WAITING; lastStateChangeTime = currentTime;
      strip.clear(); strip.show();
    } else {
      int flickerDelay = (flickerHz > 0) ? (1000 / (flickerHz * 2)) : 1000;
      if (currentTime - lastFlickerToggleTime >= flickerDelay) {
        lastFlickerToggleTime = currentTime; isLedOn = !isLedOn;
        const int* currentGroup = groups[currentStep];
        uint32_t color = isLedOn ? flickerColor : 0;
        for (int i = 0; i < 3; i++) strip.setPixelColor(currentGroup[i], color);
        strip.show();
      }
    }
  }
}
